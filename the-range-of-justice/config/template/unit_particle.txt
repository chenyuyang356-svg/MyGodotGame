# ==========================================================
# 单位视觉效果配置文件 (Unit Effect Configuration)
# ==========================================================

# ----------------------------------------------------------
# 1. 全局发射频率 (Emit Threshold)
# ----------------------------------------------------------
# 定义单位每移动多少距离（像素）产生一次粒子。
# -1.0: 使用默认逻辑（陆地为 1.2 * 半径，海洋为 0.5 * 半径）
# 较小的值会让粒子更密集（适合高速单位），较大的值会让粒子稀疏。
emit_threshold = 25.0

# ----------------------------------------------------------
# 2. 全局粒子缩放 (Global Particle Scale)
# ----------------------------------------------------------
# 覆盖所有该单位产生的粒子基础大小。
# -1.0: 自动根据单位半径计算 (Radius / 32.0 或 48.0)
global_particle_scale = 1.8

# ----------------------------------------------------------
# 3. 特效点配置 (Effect Points)
# ----------------------------------------------------------
# 语法: effect_point = 粒子类型, X偏移, Y偏移, 缩放覆盖, 寿命覆盖, 是否波纹
#
# 参数说明:
#   - 粒子类型: 必须匹配 "WaterRipple", "WaterFoam", 或 "Dust"
#   - X偏移: 左右位置。正数为右侧，负数为左侧。
#   - Y偏移: 前后位置。正数为前方（车头/船头），负数为后方（车尾/船尾）。
#   - 缩放覆盖: 该点的特定缩放。输入 -1 则使用 global_particle_scale 或默认值。
#   - 寿命覆盖: 粒子存活秒数。输入 -1 则使用硬编码默认值（泡沫2.0s, 波纹1.5s, 尘土0.75s）。
#   - 是否波纹: true 或 false。影响 Shader 的渲染模式（如环形扩散效果）。
#
# 特殊逻辑:
#   - 如果你定义了任何一个 "WaterRipple" 点，程序将不再在默认船头位置生成波纹。
#   - 所有的点都会继承该类型硬编码的【初速度】和【高度偏移】（例如 Dust 会有向后的随机力）。
# ----------------------------------------------------------

# --- 案例 A: 舰船双尾迹配置 ---
# 在船尾左右两侧对称生成泡沫，而不是只在正中心生成
effect_point = WaterFoam, -12.0, -35.0, -1, 2.5, false
effect_point = WaterFoam,  12.0, -35.0, -1, 2.5, false

# 在船头更靠前的位置生成一圈巨大的波纹
effect_point = WaterRipple, 0.0, 50.0, 3.5, 2.0, true

# --- 案例 B: 坦克双履带尘土配置 (如果这是个坦克单位) ---
# 坦克通常有两条履带，我们在履带末端分别设置出烟点

# effect_point = Dust,  15.0, -20.0, 0.8, -1, false
